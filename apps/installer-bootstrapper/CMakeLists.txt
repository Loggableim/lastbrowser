cmake_minimum_required(VERSION 3.25)
project(LastbrowserBootstrapper LANGUAGES CXX RC)

add_executable(LastbrowserBootstrapper WIN32 bootstrapper.cpp bootstrapper_core.cpp bootstrapper.rc)
target_compile_features(LastbrowserBootstrapper PRIVATE cxx_std_17)
target_compile_definitions(LastbrowserBootstrapper PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_include_directories(LastbrowserBootstrapper PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_property(TARGET LastbrowserBootstrapper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_link_libraries(LastbrowserBootstrapper PRIVATE winhttp crypt32 wintrust bcrypt gdiplus shell32 advapi32 ole32 comctl32 user32 gdi32 version)
if(MSVC)
  target_compile_options(LastbrowserBootstrapper PRIVATE /W4 /WX /EHsc /permissive- /utf-8)
  target_link_options(LastbrowserBootstrapper PRIVATE /MANIFEST:EMBED "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
endif()

enable_testing()
add_executable(bootstrapper-core-tests tests/core_tests.cpp bootstrapper_core.cpp)
target_compile_features(bootstrapper-core-tests PRIVATE cxx_std_17)
target_include_directories(bootstrapper-core-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_property(TARGET bootstrapper-core-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_link_libraries(bootstrapper-core-tests PRIVATE winhttp)
if(MSVC)
  target_compile_options(bootstrapper-core-tests PRIVATE /W4 /WX /EHsc /permissive- /utf-8)
endif()
add_test(NAME bootstrapper-core-tests COMMAND bootstrapper-core-tests)
